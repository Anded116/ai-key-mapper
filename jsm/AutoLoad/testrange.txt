# TestRange turns 0.022° per mouse count at sensitivity 1 (Source-style).

IN_GAME_SENS = 1

##### Left stick: WASD (the game also reads the stick itself in hybrid mode)
LUP = W
LDOWN = S
LLEFT = A
LRIGHT = D

##### Shoulders and triggers
L = SPACE                       # Jump
R = LMOUSE                      # Fire
ZL = RMOUSE                     # Aim
ZR = NONE                       # Weapon chord, see Presets/TestRange

##### Face buttons
S = E                           # Use
W = R                           # Reload
N = Q                           # Last weapon
E = LCONTROL                    # Crouch
R3 = V                          # Melee

##### D-pad
UP = G                          # Grenade
DOWN = F                        # Flashlight
LEFT = SCROLLUP                 # Previous weapon
RIGHT = SCROLLDOWN              # Next weapon

##### Options / Create: tap / hold
+ = ESC M                       # Pause / map
- = I TAB                       # Inventory / stats while held
ZR,- = F5                       # Quick save
ZR,+ = F9                       # Quick load

##### Touchpad: touch = gyro off, click left / right half

GRID_SIZE = 2 1
GYRO_OFF = TOUCH
CAPTURE = NONE

T2,CAPTURE = F1                 # Help
